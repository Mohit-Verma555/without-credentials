pipeline {
    agent any
    stages
    {
        stage ("Pull") {
            steps {
                git 'https://github.com/Mohit-Verma555/sonar.git'
            }
        }
        stage ("build") {
            steps {
                sh '/opt/maven/bin/mvn clean package'
            }
        }
        stage ("test")
        {
            steps {
			withSonarQubeEnv(installationName: 'sonar-install', credentialsId: 'sonar-id') {
                       sh ''' /opt/maven/bin/mvn clean verify org.sonarsource.scanner.maven:sonar-maven-plugin:sonar \
                      -Dsonar.projectKey=new-pipe \
                      -Dsonar.projectName='new-pipe' '''
}
            }
        }
        stage ("Quality-gate") 
{
            steps {
                   timeout(time: 10, unit: 'SECONDS') {
                     waitForQualityGate abortPipeline: true, credentialsId: 'sonar-id'
            }
       }
}
        stage ("deploy")
        {
            steps {
                deploy adapters: [tomcat9(alternativeDeploymentContext: '', credentialsId: 'tomcat', path: '', url: 'http://16.178.128.156:8080/')], contextPath: 'student', war: '**/*.war'
            }
        }
    }
}
